using System.Collections;
using Larje.ProjectSetup.Editor.Playtest;
using NUnit.Framework;
using UnityEngine.TestTools;

namespace {{Namespace}}.Playtests
{
    public sealed class SmokePlaytest
    {
        [UnityTest]
        public IEnumerator BootsToMenu()
        {
            LogAssert.ignoreFailingMessages = true;
            using (PlaytestSession session = new PlaytestSession("smoke"))
            {
                yield return session.LoadFirstScene();
                yield return session.WaitSeconds(1f);
                yield return session.CaptureAsync("splash");
                yield return session.WaitForScene("Main");
                yield return session.WaitSeconds(2f);
                yield return session.CaptureAsync("menu");
                session.DescribeButtons();
                session.WriteReport();
                Assert.IsEmpty(session.Errors, string.Join("\n", session.Errors));
            }
        }
    }
}
